# AI 代码率统计（ai-stats）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 `.cursor/hooks/ai-stats/` 用 3 个业务文件实现 AI 代码率采集、commit trailer 对账与离线汇总（PRD V1.1 + design spec）。

**Architecture:** `core.cjs` 承载 hash/LCS/trace/归因/trailer；`hooks.cjs` 分发 Cursor/Git 入口；`ai-stats.js` 只读 Git trailer 出报表。采集只记 Agent 工具调用；对账用 hash 交集 + 同线 del/add 整对算 AI。Tab 编辑不采集（无法区分 AI 建议与手动输入，宁漏勿滥）。

**Tech Stack:** Node.js 18+、Git 2.x、Cursor Hooks（preToolUse/postToolUse）；stdin 解析内置于 `core.cjs`（仓级 `hook-stdin.cjs` / `block-destructive-ops.cjs` 已移除）

## 编码与转码规范（必做）

Windows + 中文路径 + Cursor Hook JSON 是乱码高发区，**所有 I/O 统一走封装函数**，禁止裸 `readFileSync`/`execSync` 无编码。

### 规则

| 场景 | 要求 |
|------|------|
| 文件读写 | `readTextUtf8(fp)` / `writeTextUtf8(fp, text)`，默认 `{ encoding: 'utf8' }` |
| BOM | 读入后 `stripBom()`（`\uFEFF`）；写出 **不带 BOM** |
| 换行 | `normalizeLine`：CRLF/LF 统一；Git diff / trace 对齐 |
| Hook stdin JSON | **`core.cjs` 自包含** `parseHookStdinJson`（含 Windows 路径 `\` 转义修复；不再依赖仓级 `hook-stdin.cjs`） |
| `git` 子进程 | `execGit(cmd)` 固定 `{ encoding: 'utf8', env: { ...process.env, LC_ALL: 'C.UTF-8' } }`；Windows 可设 `PYTHONUTF8=1` |
| commit message 文件 | `prepare-commit-msg` 读写 message 文件用 UTF-8，写回前 `normalizeMessageNewlines` |
| jsonl trace | 每行 `JSON.stringify` + `\n`；读行 `JSON.parse(stripBom(line))` |
| 中文内容行 | hash 前必须 `normalizeLine`；测试须含中文 fixture |

### `core.cjs` 必须导出（供测试）

```javascript
function stripBom(text) {
  return String(text).replace(/^\uFEFF/, '');
}
function readTextUtf8(fp) {
  return stripBom(fs.readFileSync(fp, 'utf8'));
}
function writeTextUtf8(fp, text) {
  fs.mkdirSync(path.dirname(fp), { recursive: true });
  fs.writeFileSync(fp, text, { encoding: 'utf8' }); // 无 BOM
}
function execGit(args) {
  return execSync(`git ${args}`, {
    encoding: 'utf8',
    env: { ...process.env, LC_ALL: 'C.UTF-8' },
  });
}
```

### 编码相关测试（见 `test/unit/encoding.test.cjs`）

- 含 BOM 的 UTF-8 文件读入后 hash 与无 BOM 一致
- CRLF 行与 LF 行 hash 一致
- 中文行内容 `('商品', 100, '数码')` hash 稳定
- 模拟 Hook stdin：`{"path":"F:\\\\AI开发\\\\repo\\\\foo.ts"}` 解析不抛错

```javascript
function ensureDir(d) {
  fs.mkdirSync(d, { recursive: true });
}

let _traceOverride = null;
function setTraceDirForTest(dir) {
  _traceOverride = dir;
}
function paths() {
  const base = _traceOverride || TRACE_DIR;
  return {
    active: path.join(base, 'active.jsonl'),
    snapshots: path.join(base, 'snapshots'),
    archive: path.join(base, 'archive'),
  };
}

function appendActiveTrace(record) {
  const p = paths();
  ensureDir(p.snapshots);
  ensureDir(path.dirname(p.active));
  fs.appendFileSync(p.active, JSON.stringify(record) + '\n', 'utf8');
}

function loadActiveTraces(filesFilter) {
  const set = filesFilter ? new Set(filesFilter) : null;
  const p = paths().active;
  if (!fs.existsSync(p)) return [];
  return fs
    .readFileSync(p, 'utf8')
    .split('\n')
    .filter(Boolean)
    .map((l) => JSON.parse(l))
    .filter((r) => !set || set.has(r.file));
}

function snapshotKey(file) {
  return crypto.createHash('sha256').update(file).digest('hex') + '.txt';
}

function writeSnapshot(file, text) {
  const p = paths();
  ensureDir(p.snapshots);
  fs.writeFileSync(path.join(p.snapshots, snapshotKey(file)), text ?? '', 'utf8');
}

function readSnapshot(file) {
  const fp = path.join(paths().snapshots, snapshotKey(file));
  return fs.existsSync(fp) ? fs.readFileSync(fp, 'utf8') : null;
}

function consumeSnapshot(file) {
  const fp = path.join(paths().snapshots, snapshotKey(file));
  if (fs.existsSync(fp)) fs.unlinkSync(fp);
}
```

- [ ] **Step 3: 实现 `attributeDiff`**

```javascript
function attributeDiff(stagedDiff, traces) {
  const byFile = new Map();
  for (const t of traces) {
    if (!byFile.has(t.file)) byFile.set(t.file, []);
    byFile.get(t.file).push(t);
  }
  let aiAdded = 0;
  let aiDeleted = 0;
  const aiFiles = new Set();

  for (const { file, hunks } of stagedDiff) {
    const records = byFile.get(file) || [];
    const addHashes = new Set();
    const delHashes = new Set();
    for (const r of records) {
      for (const ln of r.lines || []) {
        if (ln.op === 'add') addHashes.add(ln.hash);
        if (ln.op === 'del') delHashes.add(ln.hash);
      }
    }
    if (!hunks.length) continue;

    const marked = new Array(hunks.length).fill(false);
    for (let i = 0; i < hunks.length; i++) {
      const h = hunks[i];
      const hit =
        (h.type === 'add' && addHashes.has(h.hash)) ||
        (h.type === 'del' && delHashes.has(h.hash));
      if (hit) marked[i] = true;
    }
    // 规则 B：del+add 相邻对，trace 命中任一侧 → 整对 AI
    for (let i = 0; i < hunks.length - 1; i++) {
      const a = hunks[i];
      const b = hunks[i + 1];
      if (a.type === 'del' && b.type === 'add') {
        const pairHit =
          delHashes.has(a.hash) ||
          addHashes.has(a.hash) ||
          delHashes.has(b.hash) ||
          addHashes.has(b.hash);
        if (pairHit) {
          marked[i] = true;
          marked[i + 1] = true;
        }
      }
    }
    let fAdd = 0;
    let fDel = 0;
    for (let i = 0; i < hunks.length; i++) {
      if (!marked[i]) continue;
      if (hunks[i].type === 'add') fAdd++;
      else fDel++;
    }
    if (fAdd + fDel > 0) aiFiles.add(file);
    aiAdded += fAdd;
    aiDeleted += fDel;
  }
  return { aiAdded, aiDeleted, aiFiles };
```

```javascript
function parseNumstat() {
  const out = execSync('git diff --cached --numstat', { encoding: 'utf8' });
  let lines = 0;
  let files = 0;
  for (const row of out.split('\n').filter(Boolean)) {
    const [a, d] = row.split('\t');
    if (a === '-' || d === '-') continue;
    lines += Number(a) + Number(d);
    files++;
  }
  return { lines, files };
}

function parseStagedDiff() {
  const out = execSync('git diff --cached -U0 --no-color', { encoding: 'utf8' });
  const result = [];
  let current = null;
  for (const line of out.split('\n')) {
    if (line.startsWith('+++ b/')) {
      current = { file: line.slice(6), hunks: [] };
      result.push(current);
    } else if (line.startsWith('+') && !line.startsWith('+++') && current) {
      const text = line.slice(1);
      current.hunks.push({ type: 'add', line: text, hash: hashLine(text) });
    } else if (line.startsWith('-') && !line.startsWith('---') && current) {
      const text = line.slice(1);
      current.hunks.push({ type: 'del', line: text, hash: hashLine(text) });
    }
  }
  return result;
}

function buildTrailer({ authored, lines, files, aiAdded, aiDeleted, aiFiles }) {
  const rows = [
    `AI-Authored: ${authored}`,
    `Lines: ${lines}`,
    `Files: ${files}`,
  ];
  if (authored === 'yes') {
    rows.push(`AI-Lines-Added: ${aiAdded}`);
    rows.push(`AI-Lines-Deleted: ${aiDeleted}`);
    rows.push(`AI-Files: ${aiFiles}`);
    const email = process.env.CURSOR_USER_EMAIL;
    if (email) rows.push(`AI_EMAIL: ${email}`);
  }
  return rows.join('\n');
}

function injectTrailer(messagePath, trailerBlock) {
  let msg = readTextUtf8(messagePath);
  msg = msg.replace(/\nAI-Authored:[\s\S]*$/m, '').trimEnd();
  writeTextUtf8(messagePath, msg + '\n\n' + trailerBlock + '\n');
}
```

```javascript
function runPrepareCommitMsg(argv) {
  const messagePath = argv.find((a) => !a.startsWith('-'));
  if (!messagePath) return;
  try {
    const { lines, files } = parseNumstat();
    const staged = parseStagedDiff();
    const stagedFiles = staged.map((s) => s.file);
    const traces = loadActiveTraces(stagedFiles);
    const { aiAdded, aiDeleted, aiFiles } = attributeDiff(staged, traces);
    const aiTotal = aiAdded + aiDeleted;
    const authored = aiTotal > 0 ? 'yes' : 'no';
    const cappedAdd = Math.min(aiAdded, lines);
    const cappedDel = Math.min(aiDeleted, lines);
    const trailer = buildTrailer({
      authored,
      lines,
      files,
      aiAdded: authored === 'yes' ? cappedAdd : 0,
      aiDeleted: authored === 'yes' ? cappedDel : 0,
      aiFiles: aiFiles.size,
    });
    injectTrailer(messagePath, trailer);
  } catch (e) {
    const { lines, files } = parseNumstat();
    injectTrailer(messagePath, buildTrailer({ authored: 'unknown', lines, files }));
    console.error('[ai-stats] attribution failed:', e.message);
  }
}

function runPostCommit() {
  const staged = parseStagedDiff();
  const files = staged.map((s) => s.file);
  if (!files.length) return;
  const p = paths();
  ensureDir(p.archive);
  const day = new Date().toISOString().slice(0, 10);
  const archiveFile = path.join(p.archive, `${day}.jsonl`);
  const traces = loadActiveTraces(files);
  for (const t of traces) {
    fs.appendFileSync(archiveFile, JSON.stringify(t) + '\n', 'utf8');
  }
  // 从 active 移除已归档文件
  const remain = loadActiveTraces(null).filter((t) => !files.includes(t.file));
  fs.writeFileSync(p.active, remain.map((t) => JSON.stringify(t)).join('\n') + (remain.length ? '\n' : ''), 'utf8');
  for (const f of files) consumeSnapshot(f);
}
```

```javascript
function relPath(raw, roots) {
  const abs = path.resolve(raw);
  for (const root of roots) {
    const rel = path.relative(root, abs);
    if (!rel.startsWith('..') && !path.isAbsolute(rel)) return rel.replace(/\\/g, '/');
  }
  return abs.replace(/\\/g, '/');
}

function linesFromDiffOps(ops) {
  return ops
    .map(({ op, line }) => {
      const hash = hashLine(line);
      return hash ? { op, hash } : null;
    })
    .filter(Boolean);
}

function runCursorHook(input) {
  const event = input.hook_event_name || '';
  const tool = input.tool_name || '';
  const toolInput = input.tool_input || {};
  const roots = input.workspace_roots?.length
    ? input.workspace_roots.map((r) => path.resolve(String(r).replace(/^\/([a-zA-Z]):\//, '$1:/')))
    : [process.cwd()];
  const rawFile = toolInput.path || toolInput.file_path || toolInput.target_file || '';
  const file = relPath(rawFile, roots);
  if (!file) return;

  if (/preToolUse/i.test(event) && /write|strreplace/i.test(tool)) {
    let baseline = readSnapshot(file);
    if (baseline == null && fs.existsSync(path.resolve(roots[0], file))) {
      baseline = fs.readFileSync(path.resolve(roots[0], file), 'utf8');
    }
    writeSnapshot(file, baseline ?? '');
    return;
  }

  if (/postToolUse|afterTabFileEdit/i.test(event)) {
    const source = /tab/i.test(event) ? 'tab' : 'agent';
    let newText = null;
    const abs = path.resolve(roots[0], file);
    if (fs.existsSync(abs)) newText = fs.readFileSync(abs, 'utf8');
    const oldText = readSnapshot(file);
    let lines = [];
    if (oldText != null && newText != null) {
      lines = linesFromDiffOps(diffLineOps(oldText.split('\n'), newText.split('\n')));
    }
    if (lines.length) {
      appendActiveTrace({
        v: 1,
        ts: new Date().toISOString(),
        source,
        tool,
        file,
        lines,
      });
    }
    consumeSnapshot(file);
  }
}
```

```javascript
function gitLog(since) {
  const fmt = '%H%x1f%s%x1f%ae%x1f%ad%x1e';
  const out = execSync(
    `git log --since="${since}" --format=${fmt} --date=short`,
    { encoding: 'utf8', maxBuffer: 10 * 1024 * 1024 }

  return out.split('\x1e').filter(Boolean).map((chunk) => {
    const [hash, subject, email, date] = chunk.split('\x1f');
    return { hash, subject, email, date };
  });
}
```
